Sem alterar o CSS original do projeto, o ajuste deve ser feito somente no JS:

function renderHome() {
  const container = document.getElementById("sections-container");
  container.innerHTML = "";

  // colunas/seções limitadas a 5
  const visibleCategories = categories.slice(0, 5);

  visibleCategories.forEach((cat) => {
    const section = document.createElement("div");
    section.className = "section";

    const carousel = document.createElement("div");
    carousel.className = "carousel";

    const items = catalogByCategory[cat.id] || [];
    const shuffledItems = shuffleArray(items);

    // cards do carrossel reduzidos para 5
    const visibleItems = shuffledItems.slice(0, 5);

    visibleItems.forEach((item) => {
      const card = document.createElement("div");
      card.className = "card";
      carousel.appendChild(card);
    });

    section.appendChild(carousel);
    container.appendChild(section);
  });
}
